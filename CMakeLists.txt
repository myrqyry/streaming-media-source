cmake_minimum_required(VERSION 3.22)
project(obs-streaming-media VERSION 0.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_path(OBS_WEBSOCKET_INCLUDE_DIR obs-websocket-api.h PATH_SUFFIXES obs REQUIRED)
add_library(stream-resolver STATIC src/resolver.cpp src/media-queue.cpp src/playlist-metadata.cpp)
set_target_properties(stream-resolver PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(stream-resolver PUBLIC Qt6::Core)
target_include_directories(stream-resolver PUBLIC src)
add_library(runtime-update STATIC src/runtime-update.cpp)
set_target_properties(runtime-update PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(runtime-update PUBLIC Qt6::Core Qt6::Network)
target_include_directories(runtime-update PUBLIC src)
add_library(obs-streaming-media MODULE src/plugin.cpp src/media-dock.cpp src/media-websocket.cpp src/media-runtime.cpp)
target_include_directories(obs-streaming-media PRIVATE ${OBS_WEBSOCKET_INCLUDE_DIR})
target_link_libraries(obs-streaming-media PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network stream-resolver runtime-update)
set_target_properties(obs-streaming-media PROPERTIES PREFIX "")
install(TARGETS obs-streaming-media LIBRARY DESTINATION bin/64bit)
set(STREAMING_MEDIA_RUNTIME_DIR "" CACHE PATH "Prepared standalone media runtime directory")
if(STREAMING_MEDIA_RUNTIME_DIR)
  install(PROGRAMS "${STREAMING_MEDIA_RUNTIME_DIR}/yt-dlp" "${STREAMING_MEDIA_RUNTIME_DIR}/deno" DESTINATION data/runtime)
  install(FILES "${STREAMING_MEDIA_RUNTIME_DIR}/RUNTIME-NOTICE.txt" "${STREAMING_MEDIA_RUNTIME_DIR}/yt-dlp-version.txt" DESTINATION data/runtime)
  install(DIRECTORY third-party/ DESTINATION data/runtime/licenses)
endif()
enable_testing()
add_executable(resolver-test tests/resolver-test.cpp)
target_link_libraries(resolver-test PRIVATE stream-resolver)
add_test(NAME resolver-test COMMAND resolver-test)
add_executable(obs-youtube-smoke tests/obs-youtube-smoke.cpp src/media-dock.cpp)
target_link_libraries(obs-youtube-smoke PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network stream-resolver runtime-update)
add_executable(queue-test tests/queue-test.cpp)
target_link_libraries(queue-test PRIVATE stream-resolver)
add_test(NAME queue-test COMMAND queue-test)
add_executable(dock-theme-test tests/dock-theme-test.cpp src/media-dock.cpp)
target_link_libraries(dock-theme-test PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network stream-resolver runtime-update)
add_test(NAME dock-theme-test COMMAND dock-theme-test)
set_tests_properties(dock-theme-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(audio-handoff-test tests/audio-handoff-test.cpp)
target_include_directories(audio-handoff-test PRIVATE src)
target_link_libraries(audio-handoff-test PRIVATE OBS::libobs)
add_test(NAME audio-handoff-test COMMAND audio-handoff-test)
set_tests_properties(audio-handoff-test PROPERTIES TIMEOUT 10)
add_executable(recovery-test tests/recovery-test.cpp)
target_include_directories(recovery-test PRIVATE src)
add_test(NAME recovery-test COMMAND recovery-test)
add_executable(playlist-test tests/playlist-test.cpp)
target_link_libraries(playlist-test PRIVATE stream-resolver)
add_test(NAME playlist-test COMMAND playlist-test)
add_executable(runtime-test tests/runtime-test.cpp src/media-runtime.cpp)
target_link_libraries(runtime-test PRIVATE stream-resolver Qt6::Network)
add_test(NAME runtime-test COMMAND runtime-test)
add_executable(runtime-update-test tests/runtime-update-test.cpp)
target_link_libraries(runtime-update-test PRIVATE runtime-update)
target_include_directories(runtime-update-test PRIVATE src)
add_test(NAME runtime-update-test COMMAND runtime-update-test)

  set(CPACK_GENERATOR TGZ)
  set(CPACK_PACKAGE_NAME "streaming-media-source")
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  if(STREAMING_MEDIA_RUNTIME_DIR)
    set(CPACK_PACKAGE_FILE_NAME "streaming-media-source-${PROJECT_VERSION}-linux-x86_64-bundled")
  else()
    set(CPACK_PACKAGE_FILE_NAME "streaming-media-source-${PROJECT_VERSION}-linux-x86_64-system-runtime")
  endif()
  set(CPACK_PACKAGING_INSTALL_PREFIX "/obs-streaming-media")
  set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
  set(CPACK_PACKAGE_CHECKSUM SHA256)
  include(CPack)
