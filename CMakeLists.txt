cmake_minimum_required(VERSION 3.22)
project(obs-streaming-media VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(libobs REQUIRED)
add_library(stream-resolver STATIC src/resolver.cpp)
set_target_properties(stream-resolver PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(stream-resolver PUBLIC Qt6::Core)
target_include_directories(stream-resolver PUBLIC src)
add_library(obs-streaming-media MODULE src/plugin.cpp)
target_link_libraries(obs-streaming-media PRIVATE OBS::libobs stream-resolver)
set_target_properties(obs-streaming-media PROPERTIES PREFIX "")
install(TARGETS obs-streaming-media LIBRARY DESTINATION bin/64bit)
enable_testing()
add_executable(resolver-test tests/resolver-test.cpp)
target_link_libraries(resolver-test PRIVATE stream-resolver)
add_test(NAME resolver-test COMMAND resolver-test)
add_executable(obs-youtube-smoke tests/obs-youtube-smoke.cpp)
target_link_libraries(obs-youtube-smoke PRIVATE OBS::libobs Qt6::Core)
