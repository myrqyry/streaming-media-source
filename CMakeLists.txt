cmake_minimum_required(VERSION 3.22)
project(obs-streaming-media VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_path(OBS_WEBSOCKET_INCLUDE_DIR obs-websocket-api.h PATH_SUFFIXES obs REQUIRED)
add_library(stream-resolver STATIC src/resolver.cpp src/media-queue.cpp)
set_target_properties(stream-resolver PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(stream-resolver PUBLIC Qt6::Core)
target_include_directories(stream-resolver PUBLIC src)
add_library(obs-streaming-media MODULE src/plugin.cpp src/media-dock.cpp src/media-websocket.cpp)
target_include_directories(obs-streaming-media PRIVATE ${OBS_WEBSOCKET_INCLUDE_DIR})
target_link_libraries(obs-streaming-media PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network stream-resolver)
set_target_properties(obs-streaming-media PROPERTIES PREFIX "")
install(TARGETS obs-streaming-media LIBRARY DESTINATION bin/64bit)
enable_testing()
add_executable(resolver-test tests/resolver-test.cpp)
target_link_libraries(resolver-test PRIVATE stream-resolver)
add_test(NAME resolver-test COMMAND resolver-test)
add_executable(obs-youtube-smoke tests/obs-youtube-smoke.cpp src/media-dock.cpp)
target_link_libraries(obs-youtube-smoke PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network stream-resolver)
add_executable(queue-test tests/queue-test.cpp)
target_link_libraries(queue-test PRIVATE stream-resolver)
add_test(NAME queue-test COMMAND queue-test)
add_executable(dock-theme-test tests/dock-theme-test.cpp src/media-dock.cpp)
target_link_libraries(dock-theme-test PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network stream-resolver)
add_test(NAME dock-theme-test COMMAND dock-theme-test)
set_tests_properties(dock-theme-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(audio-handoff-test tests/audio-handoff-test.cpp)
target_include_directories(audio-handoff-test PRIVATE src)
target_link_libraries(audio-handoff-test PRIVATE OBS::libobs)
add_test(NAME audio-handoff-test COMMAND audio-handoff-test)
set_tests_properties(audio-handoff-test PROPERTIES TIMEOUT 10)
add_executable(recovery-test tests/recovery-test.cpp)
target_include_directories(recovery-test PRIVATE src)
add_test(NAME recovery-test COMMAND recovery-test)
